auditBugs:
  - title: "AI Context Quality vs. Response Quality Correlation"
    severity: "Design Consideration"
    file: "services/ai.service.ts — buildContextObject()"
    issue: "Quality of AI output is directly proportional to context preparation. Missing knowledge base articles produce generic AI suggestions."
    impact: "High AI rejection rates and reduced agent trust if onboarding does not enforce minimum KB count."
    fix: "Implement onboarding flow that requires minimum 5 KB articles and tone selection before AI features unlock."
  - title: "Ticket Creation Async AI Failure Masking"
    severity: "Medium Priority"
    file: "services/ticket.service.ts — createTicket()"
    issue: "Steps 1–4 of ticket creation are synchronous. Post-commit AI pipeline failure masks AI metadata silently."
    impact: "Agents opening AI-less tickets see empty AI panels without explanation."
    fix: "Add an ai_processing_status field to tickets table (pending / complete / failed) with retry option."
  - title: "Supabase Realtime Subscription Memory Management"
    severity: "Low Priority"
    file: "hooks/useTicketRealtime.ts"
    issue: "Supabase Realtime channel subscriptions must unsubscribe on component unmount."
    impact: "Memory leaks and duplicate state updates in long browser sessions."
    fix: "Verify cleanup in useEffect return: supabase.channel(channelName).unsubscribe()."

    severity: "Medium Priority"
    file: "services/ai.service.ts — loadKnowledgeBase()"

    impact: "Degraded prompt quality for businesses with 100+ KB articles."
